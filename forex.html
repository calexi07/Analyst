// Daily Observer — receives liquidity / invalidation / fractal-break / retest alerts from TradingView and stores them. // The site turns the stored signals into setups (Forming → Formed → At entry zone). // // Webhook URL: https://.supabase.co/functions/v1/tv-signal?key= (same secret as tv-daily) // Settings: turn OFF "Verify JWT" for this function. // // Body: one JSON event or a JSON array of events — optionally preceded by readable text for the phone: // {"kind":"liq","side":"sell","pair":"EURUSD","tf":"60","level":1.1620,"price":1.1631,"extreme":1.1612} // {"kind":"liqx","side":"sell","pair":"EURUSD","tf":"60","level":1.1620,"price":1.1601} ← swept level broken by a body close // {"kind":"bos","pair":"{{ticker}}","tf":"{{interval}}","price":{{close}}} // {"kind":"retest","pair":"{{ticker}}","tf":"{{interval}}","price":{{close}}} // Indicator with a fixed alert() text you can't edit? Put the context in the URL instead: // …/tv-signal?key=&kind=bos&pair=EURUSD&tf=60 — the text is stored and scanned for bull/bear. // Heartbeat (DO Heartbeat indicator) — closed candles for every pair, stored in do_candles: // {"kind":"ohlc","tf":"60","t":1760086800000,"c":[["EURUSD",o,h,l,c,fractalHigh,fractalLow], ...]} import { createClient } from "jsr:@supabase/supabase-js@2"; const cors = { "access-control-allow-origin": "*", "access-control-allow-headers": "content-type", "access-control-allow-methods": "POST, OPTIONS" }; const json = (data: unknown, status = 200) => new Response(JSON.stringify(data), { status, headers: { ...cors, "content-type": "application/json" } }); const TF: Record = { "1": "1m", "3": "3m", "5": "5m", "15": "15m", "30": "30m", "45": "45m", "60": "1H", "120": "2H", "180": "3H", "240": "4H", "480": "8H", "960": "16H", "D": "1D", "1D": "1D", "2D": "2D", "3D": "3D", "W": "1W", "1W": "1W", "M": "1M", "1M": "1M" }; const num = (v: unknown) => { if (v === null || v === undefined || v === "") return null; const n = Number(v); return Number.isFinite(n) ? n : null; }; const KINDS = ["liq", "liqx", "bos", "retest"]; function toRow(b: Record) { const kind = String(b.kind ?? b.type ?? "").toLowerCase(); if (!KINDS.includes(kind)) return { error: "kind must be liq, liqx, bos or retest" }; const plain = typeof b.text === "string" ? b.text : ""; let pairRaw = String(b.pair ?? b.ticker ?? ""); if (!pairRaw && plain) { const m = plain.toUpperCase().match(/\b(EUR|GBP|AUD|NZD|USD|CAD|CHF|JPY|XAU)(USD|JPY|CHF|CAD|GBP|AUD|NZD)\b/); if (m) pairRaw = m[0]; } const pair = pairRaw.toUpperCase().replace(/^.*:/, "").replace(/[^A-Z]/g, "").slice(0, 6); if (pair.length !== 6) return { error: "missing pair" }; const rawTf = String(b.tf ?? b.interval ?? "").toUpperCase().trim(); const tf = TF[rawTf] ?? (rawTf || "?"); let side = b.side ? String(b.side).toLowerCase() : null; if (side && ["low", "ssl", "sellside", "sell-side", "bullish"].includes(side)) side = kind === "bos" ? "bull" : "sell"; if (side && ["high", "bsl", "buyside", "buy-side", "bearish"].includes(side)) side = kind === "bos" ? "bear" : "buy"; const level = num(b.level), price = num(b.price ?? b.close), extreme = num(b.extreme); if (kind === "liq" && !side && level !== null && price !== null) side = price > level ? "sell" : "buy"; if (!side && plain) { /* read the direction from an indicator's own alert text */ const t = plain.toLowerCase(); if (kind === "bos") side = /bear|down|short|sell|lower/.test(t) ? "bear" : /bull|up\b|long|buy|higher/.test(t) ? "bull" : null; if (kind === "liq" || kind === "liqx") side = /sell[- ]?side|ssl|\blow/.test(t) ? "sell" : /buy[- ]?side|bsl|\bhigh/.test(t) ? "buy" : null; } let at = new Date(); const ta = b.at ?? b.time; if (ta !== undefined && ta !== null && ta !== "") { const d = /^\d{10,13}$/.test(String(ta)) ? new Date(Number(ta) * (String(ta).length > 11 ? 1 : 1000)) : new Date(String(ta)); if (!isNaN(d.getTime())) at = d; } return { row: { pair, tf, kind, side, level, price, extreme, at: at.toISOString(), source: String(b.source ?? "tv"), raw: b } }; } /* heartbeat: one closed candle per pair → do_candles (upsert, so a repeated alert changes nothing) */ async function storeCandles(b: Record) { const rawTf = String(b.tf ?? "").toUpperCase().trim(), tf = TF[rawTf] ?? rawTf; const t0 = num(b.t), list = Array.isArray(b.c) ? b.c : []; const rows = []; for (const r of list) { if (!Array.isArray(r) || r.length < 5) continue; const pair = String(r[0] ?? "").toUpperCase().replace(/^.*:/, "").replace(/[^A-Z]/g, ""); const hasT = r.length >= 8, k = hasT ? 2 : 1, t = hasT ? num(r[1]) : t0; /* row may carry its own time */ const [o, h, l, c, fh, fl] = [r[k], r[k + 1], r[k + 2], r[k + 3], r[k + 4], r[k + 5]].map(num); if (pair.length !== 6 || t === null || c === null) continue; rows.push({ pair, tf, t: new Date(t).toISOString(), o, h, l, c, fh, fl }); } if (!rows.length) return json({ ok: false, error: "no candles in message" }, 400); const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!, { auth: { persistSession: false } }); const { error } = await sb.from("do_candles").upsert(rows, { onConflict: "pair,tf,t" }); if (error) return json({ ok: false, error: error.message }, 500); await sb.from("do_settings").upsert({ key: "hb_" + tf, value: { at: new Date().toISOString(), t: rows[0].t, pairs: rows.length } }, { onConflict: "key" }); return json({ ok: true, tf, t: rows[0].t, candles: rows.length }); } Deno.serve(async (req) => { if (req.method === "OPTIONS") return new Response(null, { headers: cors }); if (req.method !== "POST") return json({ ok: true, hint: "POST indicator alerts here" }); const url = new URL(req.url); const secret = Deno.env.get("TV_SECRET"); if (!secret || url.searchParams.get("key") !== secret) return json({ ok: false, error: "forbidden" }, 403); /* defaults from the webhook URL (?kind=bos&pair=EURUSD&tf=60) — for indicators whose text you can't edit */ const q: Record = {}; for (const k of ["kind", "pair", "tf", "side", "level"]) { const v = url.searchParams.get(k); if (v) q[k] = v; } const text = await req.text(); /* the body may be pure JSON, or friendly text for the phone followed by the JSON data */ let parsed: unknown = null; try { parsed = JSON.parse(text); } catch { const i = text.search(/[\[{]/); if (i >= 0) for (const end of [text.lastIndexOf("]"), text.lastIndexOf("}")].sort((x, y) => y - x)) { if (end > i) { try { parsed = JSON.parse(text.slice(i, end + 1)); break; } catch { /* keep trying */ } } } if (parsed === null) parsed = { text }; } if (parsed && typeof parsed === "object" && !Array.isArray(parsed) && String((parsed as Record).kind).toLowerCase() === "ohlc") return storeCandles(parsed as Record); const items = (Array.isArray(parsed) ? parsed : [parsed]).map((x) => ({ ...q, ...(x && typeof x === "object" ? x as Record : { text }) })); const rows = [], errors = []; for (const it of items) { const r = toRow(it); if ("row" in r) rows.push(r.row); else errors.push(r.error); } if (!rows.length) return json({ ok: false, error: errors[0] ?? "nothing to store", got: text.slice(0, 200) }, 400); const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!, { auth: { persistSession: false } }); const { error } = await sb.from("do_signals").insert(rows); if (error) return json({ ok: false, error: error.message }, 500); return json({ ok: true, stored: rows.map((r) => ({ pair: r.pair, tf: r.tf, kind: r.kind, side: r.side, level: r.level })), skipped: errors }); });
